    180 _vin_git_modified() {
    181   local repo_root files file rel
    182   repo_root=$(git rev-parse --show-toplevel 2>/dev/null) || return
    183   files=(${(f)"$(git status --porcelain --untracked-files=no 2>/dev/null | awk '/^ M/{print $2}')"})
    184   for i in {1..$#files}; do
    185     file="$repo_root/${files[$i]}"
    186     rel=${file:#$PWD/}
    187     files[$i]="${rel#$PWD/}"
    188   done
    189   compadd -U -a files
    190 }
    191 compdef _vin_git_modified vin
