     12 " format folded section header text
     13 function! GenericFoldText()
     14     let line = getline(v:foldstart)
     15 
     16     let nucolwidth = &fdc + &number * &numberwidth
     17     let windowwidth = winwidth(0) - nucolwidth - 3
     18     let foldedlinecount = v:foldend - v:foldstart
     19 
     20     let onetab = strpart('          ', 0, &tabstop)
     21     let line = substitute(line, '\t', onetab, 'g')
     22 
     23     let line = strpart(line, 0, windowwidth - len(foldedlinecount))
     24     let level = (len(split(line)[0]) - 1) * 3
     25     let level_indent = repeat(" ", level)
     26     let fillcharcount = windowwidth - len(line) - len(foldedlinecount) - level
     27     return level_indent . line . repeat(" ",fillcharcount) . ' ' . foldedlinecount
     28 endfunction
     29 setlocal foldtext=GenericFoldText()
