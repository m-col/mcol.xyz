     23 void assign_colour(char *hex, float dest[4]) {
     24     if (!hex) {
     25 	return;
     26     }
     27     if (hex[0] == '#') {
     28 	hex++;
     29     }
     30 
     31     char c[2];
     32     int copy;
     33     if (strlen(hex) == 6) {
     34 	dest[3] = 1;
     35 	copy = 3;
     36     } else if (strlen(hex) == 8) {
     37 	copy = 4;
     38     } else {
     39 	wlr_log(WLR_INFO, "Invalid colour: %s. Should be in the form #rrggbb.", hex);
     40 	return;
     41     }
     42 
     43     for (int i = 0; i < copy; i++) {
     44 	strncpy(c, &hex[i * 2], 2);
     45 	dest[i] = (float)strtol(c, NULL, 16) / 255;
     46     }
     47 }

     50 static void load_wallpaper(struct desk *desk, char *path) {
     51     cairo_surface_t *image = cairo_image_surface_create_from_png(path);
     52 
     53     if (!image || cairo_surface_status(image) != CAIRO_STATUS_SUCCESS) {
     54 	wlr_log(WLR_INFO, "Could not load image: %s", path);
     55 	return;
     56     }
     57 
     58     struct wallpaper *wallpaper = calloc(1, sizeof(struct wallpaper));
     59     wallpaper->width = cairo_image_surface_get_width(image);
     60     wallpaper->height = cairo_image_surface_get_height(image);
     61     int stride = cairo_format_stride_for_width(CAIRO_FORMAT_ARGB32, wallpaper->width);
     62 
     63     // we copy the image to a second surface to ensure the pixel format is
     64     // compatible with wlr_texture_from_pixels
     65     cairo_surface_t *canvas = cairo_image_surface_create(
     66 	CAIRO_FORMAT_ARGB32, wallpaper->width, wallpaper->height);
     67     cairo_t *cr = cairo_create(canvas);
     68     cairo_set_source_surface(cr, image, 0, 0);
     69     cairo_paint(cr);
     70 
     71     wallpaper->texture = wlr_texture_from_pixels(
     72 	wimp.renderer, DRM_FORMAT_ARGB8888, stride, wallpaper->width,
     73 	wallpaper->height, cairo_image_surface_get_data(canvas)
     74     );
     75     desk->wallpaper = wallpaper;
     76     damage_all_outputs();
     77     cairo_destroy(cr);
     78     cairo_surface_destroy(image);
     79     cairo_surface_destroy(canvas);
     80 }

     83 static void set_wallpaper(struct desk *desk, char *wallpaper) {
     84     // wallpaper is a colour
     85     if (strlen(wallpaper) == 7 && wallpaper[0] == '#') {
     86 	assign_colour(wallpaper, desk->background);
     87 	return;
     88     }
     89 
     90     // wallpaper is a path to an image
     91     wordexp_t p;
     92     wordexp(wallpaper, &p, WRDE_NOCMD | WRDE_UNDEF);
     93     load_wallpaper(desk, p.we_wordv[0]);
     94     wordfree(&p);
     95 }

    114 	XKB_KEY_F5,

    121 	kb->mods = WLR_MODIFIER_CTRL;

    126 	wl_list_insert(wimp.key_bindings.prev, &kb->link);

    141 	    configure_desks(strtod(s, NULL));

    149 	    s = strtok(NULL, " \t\n\r");

    151 	    bool found = false;

    160 	    }
    161

    166 		    set_wallpaper(desk, s);
    167 		}
    168 	    }
    169 
    170 	    // desk <index> borders [focus|normal] <#rrggbb>
    171 	    // desk <index> borders width <int>
    172 	    else if (!strcasecmp(s, "borders")) {
    173 		s = strtok(NULL, " \t\n\r");

    185 	    }

    187 	    // desk <index> corners [focus|normal] <#rrggbb>

    197 	}

    205     }

    212     }

    276 static int _startup(void *vdata) {
    277     struct startup_data *data = vdata;
    278     wl_event_source_remove(data->wl_event_source);
    279     if (fork() == 0) {
    280 	if (execl(data->script, data->script, (void *)NULL) == -1) {
    281 	    exit(EXIT_FAILURE);
    282 	}
    283     } else {
    284 	wlr_log(WLR_DEBUG, "Running startup script: %s", data->script);
    285     }
    286     free(data->script);
    287     free(data);
    288     return 0;
    289 }

    292 void schedule_startup() {
    293     char *xdg_config = getenv("XDG_CONFIG_HOME");
    294     char *config_dir = (xdg_config && *xdg_config) ? CONFIG_HOME_XDG : CONFIG_HOME;
    295     struct startup_data *data = calloc(1, sizeof(struct startup_data));
    296     wordexp_t p;
    297     wordexp(config_dir, &p, WRDE_NOCMD | WRDE_UNDEF);
    298     data->script = strdup(p.we_wordv[0]);
    299     wordfree(&p);
    300 
    301     if (is_executable(data->script)) {
    302 	struct wl_event_loop *event_loop = wl_display_get_event_loop(wimp.display);
    303 	data->wl_event_source = wl_event_loop_add_timer(event_loop, _startup, data);
    304 	wl_event_source_timer_update(data->wl_event_source, 1);
    305     } else {
    306 	wlr_log(WLR_ERROR, "Cannot run startup script: %s", data->script);
    307 	free(data->script);
    308 	free(data);
    309     }
