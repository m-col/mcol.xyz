    120 int main(int argc, char *argv[])
    121 {
    122     int opt;
    123     int log_level = WLR_ERROR;
    124 
    125     while ((opt = getopt(argc, argv, "hc:di")) != -1) {
    126         switch (opt) {
    127 	    case 'h':
    128 		printf(usage);
    129 		return EXIT_SUCCESS;
    130 		break;
    131 	    case 'd':
    132 		log_level = WLR_DEBUG;
    133 		break;
    134 	    case 'i':
    135 		log_level = WLR_INFO;
    136 		break;
    137         }
    138     }
    139 
    140     init_log(log_level);
    141 
    142     // create
    143     wimp.display = wl_display_create();
    144     wimp.backend = wlr_backend_autocreate(wimp.display);
    145     wimp.renderer = wlr_backend_get_renderer(wimp.backend);
    146     wlr_renderer_init_wl_display(wimp.renderer, wimp.display);
    147     wlr_compositor_create(wimp.display, wimp.renderer);
    148 
    149     const char *socket = wl_display_add_socket_auto(wimp.display);
    150     if (!socket) {
    151 	wlr_log(WLR_INFO, "Could not create socket: %s.", socket);
    152 	return EXIT_FAILURE;
    153     }
    154 
    155     // initialise
    156     wl_list_init(&wimp.desks);
    157     wl_list_init(&wimp.key_bindings);
    158     wl_list_init(&wimp.mouse_bindings);
    159     wl_list_init(&wimp.marks);
    160     wl_list_init(&wimp.scratchpads);
    161 
    162     // configure
    163     set_up_inputs();
    164     set_up_outputs();
    165     set_up_shell();
    166     set_up_cursor();
    167     set_up_decorations();
    168     set_up_layer_shell();
    169     set_up_defaults();
    170 
    171     // start
    172     if (!start_ipc(socket) || !wlr_backend_start(wimp.backend)) {
    173 	shutdown();
    174 	return EXIT_FAILURE;
    175     }
    176     setenv("WAYLAND_DISPLAY", socket, true);
    177     schedule_startup();
    178     centre_cursor();
    179     wlr_log(WLR_INFO, "Starting with WAYLAND_DISPLAY=%s.", socket);
    180     wl_display_run(wimp.display);
    181 
    182     // stop
    183     wl_display_destroy_clients(wimp.display);
    184     close_ipc(socket);
    185     shutdown();
    186     return EXIT_SUCCESS;
    187 }
