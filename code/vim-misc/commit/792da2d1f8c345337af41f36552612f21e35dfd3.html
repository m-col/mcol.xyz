+function! s:goyo_enter()
+    set noshowmode
+    set noshowcmd
+endfunction
+
+function! s:goyo_leave()
+    source ~/.vim/bundle/vim-misc/plugin/theme.vim
+endfunction
+
+autocmd! User GoyoEnter nested call <SID>goyo_enter()
