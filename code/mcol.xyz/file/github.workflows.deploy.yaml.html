      1 name: Deploy
      2 
      3 on:
      4   push:
      5     branches: [master]
      6   schedule:
      7     - cron: "0 0 * * *"
      8   workflow_dispatch:
      9 
     10 concurrency:
     11   group: pages
     12   cancel-in-progress: true
     13 
     14 permissions:
     15   contents: write
     16 
     17 jobs:
     18   deploy:
     19     runs-on: ubuntu-latest
     20     steps:
     21       - uses: actions/checkout@v4
     22         with:
     23           # gitja walks each repo's full commit history, and libgit2 (which
     24           # gitja uses) doesn't handle git's shallow-clone boundary the way
     25           # native git does - a shallow fetch here makes it error out
     26           # trying to look up commits that were never downloaded.
     27           fetch-depth: 0
     28 
     29       - name: Install uv
     30         uses: astral-sh/setup-uv@v4
     31         with:
     32           enable-cache: true
     33 
     34       - name: Build blog
     35         run: |
     36           uv sync --locked
     37           uv run pelican content -o output -s publishconf.py
     38 
     39       - uses: haskell-actions/setup@v2
     40         with:
     41           enable-stack: true
     42 
     43       - name: Checkout gitja
     44         uses: actions/checkout@v4
     45         with:
     46           repository: m-col/gitja
     47           path: gitja-src
     48           fetch-depth: 0
     49 
     50       - uses: actions/cache@v5
     51         with:
     52           path: ~/.stack
     53           key: ${{ runner.os }}-stack-${{ hashFiles('gitja-src/stack.yaml.lock') }}
     54 
     55       - name: Build gitja
     56         run: |
     57           cd gitja-src
     58           stack --no-terminal install
     59           echo "$HOME/.local/bin" >> "$GITHUB_PATH"
     60 
     61       - name: Install dhall-to-json
     62         run: |
     63           curl -fsSL \
     64             https://github.com/dhall-lang/dhall-haskell/releases/download/1.42.2/dhall-json-1.7.12-x86_64-linux.tar.bz2 \
     65             | tar xj -C /usr/local/bin --strip-components=1 bin/dhall-to-json
     66 
     67       - name: Clone repos listed on /code
     68         run: |
     69           repos=$(dhall-to-json --file gitja/config.dhall | jq -r '.repos[]' | sed 's#^\.\./##')
     70           mv gitja-src ../gitja
     71           cd ..
     72           for repo in $repos; do
     73             case "$repo" in
     74               # mcol.xyz is already checked out here as this workflow's own
     75               # repo, and gitja was just moved into place above (it was
     76               # already checked out to build the gitja binary itself).
     77               mcol.xyz | gitja) continue ;;
     78             esac
     79             # Full clone, not shallow - see the fetch-depth comment above.
     80             git clone "https://github.com/m-col/$repo" "$repo"
     81           done
     82 
     83       - name: Write repo descriptions
     84         run: python3 gitja/descriptions.py
     85 
     86       - name: Build /code pages
     87         run: gitja -c gitja/config.dhall -q
     88 
     89       - name: Assemble site
     90         run: |
     91           mkdir -p _site
     92           cp -r output/. _site/
     93           mkdir -p _site/code
     94           cp -r gitja/output/. _site/code/
     95 
     96       - name: Deploy to GitHub Pages
     97         uses: JamesIves/github-pages-deploy-action@v4
     98         with:
     99           branch: gh-pages
    100           folder: _site
    101           cname: mcol.xyz
