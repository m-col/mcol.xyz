     32 void scratchpad_apply_geo(struct scratchpad *scratchpad) {
     33     struct wlr_output *output =
     34 	wlr_output_layout_output_at(wimp.output_layout, wimp.cursor->x, wimp.cursor->y);
     35     struct wlr_box *ogeo = wlr_output_layout_get_box(wimp.output_layout, output);
     36     struct view *view = scratchpad->view;
     37     struct wlr_box *geo = &scratchpad->geo;
     38 
     39     if (geo->x > 0) {
     40 	view->x = geo->x;
     41     } else {
     42 	view->x = ogeo->x - (ogeo->width * geo->x / 100);
     43     }
     44     if (geo->y > 0) {
     45 	view->y = geo->y;
     46     } else {
     47 	view->y = ogeo->y - (ogeo->height * geo->y / 100);
     48     }
     49 
     50     struct wlr_box *sgeo = &view->surface->geometry;
     51     if (geo->width > 0) {
     52 	sgeo->width = geo->width;
     53     } else {
     54 	sgeo->width = - ogeo->width * geo->width / 100;
     55     }
     56     if (geo->height > 0) {
     57 	sgeo->height = geo->height;
     58     } else {
     59 	sgeo->height = - ogeo->height * geo->height / 100;
     60     }
     61     wlr_xdg_toplevel_set_size(view->surface, sgeo->width, sgeo->height);
     62 }
