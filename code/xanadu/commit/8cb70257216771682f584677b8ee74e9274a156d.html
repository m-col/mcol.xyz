+-- Root folder is $(XDG_DESKTOP_DIR:-$HOME/Desktop)
+getRoot :: IO FilePath
+getRoot = do
+    fromEnv <- lookupEnv "XDG_DESKTOP_DIR"
+    case fromEnv of
+        Nothing -> liftM ((<>) "/Desktop") getHomeDirectory
+        Just "" -> liftM ((<>) "/Desktop") getHomeDirectory
