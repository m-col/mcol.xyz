     82 bool get_action(
     83     char *name, action *found, char *args, void **data, char *response, int flag
     84 ) {
     85     // flag can be PINCH or SCROLL to access their actions, otherwise neither
     86     *data = NULL;
     87     size_t i;
     88 
     89     switch (flag) {
     90 	case PINCH:
     91 	    for (i = 0; i < sizeof(pinch_map) / sizeof(pinch_map[0]); i++) {
     92 		if (strcmp(pinch_map[i].name, name) == 0) {
     93 		    *found = pinch_map[i].action;
     94 		    action begin = pinch_map[i].begin;
     95 		    *data = calloc(1, sizeof(action));
     96 		    *(action *)(*data) = begin;
     97 		    return true;
     98 		}
     99 	    }
    100 	    break;
    101 
    102 	case SCROLL:
    103 	    for (i = 0; i < sizeof(scroll_map) / sizeof(scroll_map[0]); i++) {
    104 		if (strcmp(scroll_map[i].name, name) == 0) {
    105 		    *found = scroll_map[i].action;
    106 		    return true;
    107 		}
    108 	    }
    109 	    break;
    110 
    111 	default:
    112 	    for (i = 0; i < sizeof(action_map) / sizeof(action_map[0]); i++) {
    113 		if (strcmp(action_map[i].name, name) == 0) {
    114 		    *found = action_map[i].action;
    115 
    116 		    if (action_map[i].data_handler != NULL) {
    117 			if (!action_map[i].data_handler(data, args)) {
    118 			    sprintf(response, "Command malformed/incomplete.");
    119 			    return false;
    120 			}
    121 		    }
    122 
    123 		    return true;
    124 		}
    125 	    }
    126     }
    127 
    128     return false;
    129 }

    181 static void move_window(void *data) {
    182     struct view *view;
    183     if (wimp.grabbed_view) {
    184 	view = wimp.grabbed_view;
    185     } else {
    186 	double sx, sy;
    187 	struct wlr_surface *surface;
    188 	bool is_layer;
    189 	view = under_pointer(&surface, &sx, &sy, &is_layer);
    190 	if (!view || is_layer) {
    191 	    return;
    192 	}
    193     }
    194     double zoom = wimp.current_desk->zoom;
    195     struct wlr_box damage = {
    196 	.x = view->x * zoom,
    197 	.y = view->y * zoom,
    198 	.width = view->surface->geometry.width * zoom,
    199 	.height = view->surface->geometry.height * zoom,
    200     };
    201     struct motion motion = *(struct motion*)data;
    202     if (motion.dx > 0) {
    203 	damage.width += motion.dx;
    204     } else {
    205 	damage.x -= motion.dx;
    206 	damage.width -= motion.dx;
    207     }
    208     if (motion.dy > 0) {
    209 	damage.height += motion.dy;
    210     } else {
    211 	damage.y -= motion.dy;
    212 	damage.height -= motion.dy;
    213     }
    214     view->x += motion.dx;
    215     view->y += motion.dy;
    216     damage_box(&damage, true);
    217 }

    220 static void focus_in_direction(void *data) {
    221     if (wl_list_empty(&wimp.current_desk->views)) {
    222 	return;
    223     }
    224 
    225     struct view *current = wl_container_of(wimp.current_desk->views.next, current, link);
    226     enum direction dir = *(enum direction*)data;
    227 
    228     struct view *next = NULL;
    229     struct view *view;
    230     double vx, vy, dx, dy, vdist, dist;
    231     double x = current->x + current->surface->geometry.width / 2;
    232     double y = current->y + current->surface->geometry.height / 2;
    233     double c = y - x;  // y = x + c    / slope
    234     double n = y + x;  // y = n - x    \ slope
    235     bool above_c = dir & (DOWN | LEFT);
    236     bool above_n = dir & (DOWN | RIGHT);
    237 
    238     wl_list_for_each(view, &wimp.current_desk->views, link) {
    239 	vx = view->x + view->surface->geometry.width / 2;
    240 	vy = view->y + view->surface->geometry.height / 2;
    241 	dx = x - vx;
    242 	dy = y - vy;
    243 	vdist = dx * dx + dy * dy;
    244 	if (
    245 	    (above_c ^ (vy - vx < c)) &&
    246 	    (above_n ^ (vy + vx < n)) &&
    247 	    (!next || vdist < dist) &&
    248 	    view != current
    249 	) {
    250 	    next = view;
    251 	    dist = vdist;
    252 	}
    253     }
    254 
    255     if (next) {
    256 	unfullscreen();
    257 	pan_to_view(next);
    258 	focus_view(next, NULL);
    259     } else {
    260 	pan_to_view(current);
    261 	focus_view(current, NULL);
    262     }
    263 }

    288 void pan_desk(void *data) {
    289     struct desk *desk = wimp.current_desk;
    290     struct motion motion = *(struct motion*)data;
    291     double dx = motion.dx / desk->zoom;
    292     double dy = motion.dy / desk->zoom;
    293     if (motion.is_percentage) {
    294 	struct wlr_box *extents = wlr_output_layout_get_box(wimp.output_layout, NULL);
    295 	dx = extents->width * (dx / 100);
    296 	dy = extents->height * (dy / 100);
    297     }
    298     unfullscreen();
    299     struct view *view;
    300     wl_list_for_each(view, &desk->views, link) {
    301 	view->x -= dx;
    302 	view->y -= dy;
    303     }
    304     desk->panned_x -= dx;
    305     desk->panned_y -= dy;
    306     damage_all_outputs();
    307 }

    310 static void zoom(void *data) {
    311     /* Passed value (data) is the percentage step (+ve or -ve) */
    312     struct desk *desk = wimp.current_desk;
    313     double f = 1 + (*(double*)data / 100);
    314 
    315     double next_zoom = desk->zoom * f;
    316     if (next_zoom < wimp.zoom_min) {
    317 	f = wimp.zoom_min / desk->zoom;
    318     } else if (next_zoom > wimp.zoom_max) {
    319 	f = wimp.zoom_max / desk->zoom;
    320     }
    321     unfullscreen();
    322     desk->zoom *= f;
    323     double fx = wimp.cursor->x * (f - 1) / desk->zoom;
    324     double fy = wimp.cursor->y * (f - 1) / desk->zoom;
    325     struct view *view;
    326     wl_list_for_each(view, &desk->views, link) {
    327 	view->x -= fx;
    328 	view->y -= fy;
    329     }
    330     desk->panned_x -= fx;
    331     desk->panned_y -= fy;
    332     damage_all_outputs();
    333 }

    378 void actually_set_mark(const xkb_keysym_t sym) {
    379     struct mark *mark;
    380     struct mark *existing = NULL;
    381 
    382     if (sym == XKB_KEY_Escape) {
    383 	mark = wl_container_of(wimp.marks.next, mark, link);
    384 	wl_list_remove(&mark->link);
    385 	free(mark);
    386 	return;
    387     }
    388 
    389     wl_list_for_each(mark, &wimp.marks, link) {
    390 	if (mark->key == sym) {
    391 	    existing = mark;
    392 	    break;
    393 	}
    394     }
    395     if (existing) {
    396 	wl_list_remove(&existing->link);
    397 	free(existing);
    398     }
    399 
    400     mark = wl_container_of(wimp.marks.next, mark, link);
    401     mark->key = sym;
    402 
    403     if (wimp.bind_marks) {
    404 	struct binding *kb, *tmp;
    405 	wl_list_for_each_safe(kb, tmp, &wimp.key_bindings, link) {
    406 	    if (kb->key == sym && kb->mods == 0) {
    407 		return;
    408 	    }
    409 	}
    410 	kb = calloc(1, sizeof(struct binding));
    411 	kb->mods = 0;
    412 	kb->key = sym;
    413 	kb->data = calloc(1, sizeof(xkb_keysym_t));
    414 	kb->action = &actually_go_to_mark;
    415 	*(xkb_keysym_t *)(kb->data) = sym;
    416 	wl_list_insert(&wimp.key_bindings, &kb->link);
    417     }
    418 }

    427 void actually_go_to_mark(void *data) {
    428     const xkb_keysym_t sym = *(xkb_keysym_t *)data;
    429     struct mark *mark;
    430 
    431     if (sym == XKB_KEY_Escape) {
    432 	return;
    433     }
    434 
    435     bool found = false;
    436     wl_list_for_each(mark, &wimp.marks, link) {
    437 	if (mark->key == sym) {
    438 	    found = true;
    439 	    break;
    440 	}
    441     }
    442     if (!found) {
    443 	return;
    444     }
    445 
    446     unfullscreen();
    447     struct motion motion = {
    448 	.dx = mark->desk->panned_x - mark->x,
    449 	.dy = mark->desk->panned_y - mark->y,
    450 	.is_percentage = false,
    451     };
    452     mark->desk->zoom = mark->zoom;
    453     set_desk(mark->desk);
    454     pan_desk(&motion);
    455 
    456     struct view *view;
    457     struct wlr_box *extents = wlr_output_layout_get_box(wimp.output_layout, NULL);
    458     wl_list_for_each(view, &wimp.current_desk->views, link) {
    459 	if (
    460 	    view->x + view->surface->geometry.width < 0 || extents->width < view->x ||
    461 	    view->y + view->surface->geometry.height < 0 || extents->height < view->y
    462 	) {
    463 	    continue;
    464 	}
    465 	focus_view(view, NULL);
    466 	return;
    467     }
    468 }

    482 static void halfimize(void *data) {
    483     struct wlr_surface *surface = wimp.seat->keyboard_state.focused_surface;
    484     if (!surface || !wlr_surface_is_xdg_surface(surface) || wl_list_empty(&wimp.current_desk->views)) {
    485 	return;
    486     }
    487     struct view *view = wl_container_of(wimp.current_desk->views.next, view, link);
    488     enum direction dir = *(enum direction*)data;
    489     double vx = view->x + view->surface->geometry.width / 2;
    490     double vy = view->y + view->surface->geometry.height / 2;
    491     double x, y, width, height;
    492     wlr_output_layout_closest_point(wimp.output_layout, NULL, vx, vy, &x, &y);
    493     struct wlr_output *output = wlr_output_layout_output_at(wimp.output_layout, x, y);
    494     unfullscreen();
    495 
    496     switch (dir) {
    497 	case RIGHT:
    498 	    x = output->width / 2;
    499 	    y = 0;
    500 	    width = output->width / 2;
    501 	    height = output->height;
    502 	    break;
    503 	case LEFT:
    504 	    x = 0;
    505 	    y = 0;
    506 	    width = output->width / 2;
    507 	    height = output->height;
    508 	    break;
    509 	case UP:
    510 	    x = 0;
    511 	    y = 0;
    512 	    width = output->width;
    513 	    height = output->height / 2;
    514 	    break;
    515 	case DOWN:
    516 	    x = 0;
    517 	    y = output->height / 2;
    518 	    width = output->width;
    519 	    height = output->height / 2;
    520 	    break;
    521 	case NO_DIRECTION:
    522 	    return;
    523     }
    524 
    525     double zoom = wimp.current_desk->zoom;
    526     int border_width = wimp.current_desk->border_width;
    527     struct wlr_box new = {
    528 	.x = x / zoom + border_width,
    529 	.y = y / zoom + border_width,
    530 	.width = (width - border_width * 2) / zoom,
    531 	.height = (height - border_width * 2) / zoom,
    532     };
    533     view_apply_geometry(view, &new);
    534 }

    598 static void to_region(void *data) {
    599     struct wlr_box *box = (struct wlr_box *)data;
    600     struct wlr_box *extents = wlr_output_layout_get_box(wimp.output_layout, NULL);
    601 
    602     struct motion motion = {
    603 	.dx = - (extents->width / 2) + (box->width / 2) + box->x,
    604 	.dy = - (extents->height / 2) + (box->height / 2) + box->y,
    605 	.is_percentage = false,
    606     };
    607     pan_desk(&motion);
    608 
    609     double f;
    610     if (extents->width / extents->height > box->width / box->height) {
    611 	f = wimp.current_desk->zoom / ((double)box->width / (double)extents->width);
    612     } else {
