+" toggle [ ] tick box on current line with ctrl-space
+function! ToggleTodo()
+    let l:line = getline('.')
+    if match(l:line, '\[x\]') != -1
+	let l:line = substitute(l:line, '\[x\]', '\[ \]', '')
+	call setline('.', l:line)
+    elseif match(l:line, '\[ \]') != -1
+	let l:line = substitute(l:line, '\[ \]', '\[x\]', '')
+	call setline('.', l:line)
+    endif
+endfunction
+nnoremap <silent> <C-@> :call ToggleTodo()<CR>
