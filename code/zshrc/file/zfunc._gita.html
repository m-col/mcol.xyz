     19 __gita_get_infos() {
     20     local -a all_infos infos_in_used infos_unused
     21     all_infos=($(_call_program commands gita info ll | cut -d ":" -f2))
     22     infos_in_used=($(echo ${all_infos[1]} | tr ',' ' '))
     23     infos_unused=($(echo ${all_infos[2]} | tr ',' ' '))
     24     _describe -t infos_used 'gita infos in used' infos_in_used
     25     _describe -t infos_unused 'gita infos unused' infos_unused
     26 }

    142 _gita_flags() {

    144     local tmp ret=1
    145     _arguments -A \
