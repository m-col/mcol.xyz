      1 #!/usr/bin/nft -f
      2 #
      3 # Restricted nftables configuration:
      4 #   - Limit inbound traffic
      5 #   - Allow outbound traffic to VPN's socks5 proxy
      6 #   - Allow outbound traffic from UID 'tor'
      7 #
      8 
      9 
     10 #
     11 # Definitions
     12 #
     13 
     14 # VPN server
     15 define vpn_ip = 123.456.789.123
     16 
     17 # individual IPs to be unrestricted
     18 define exceptions = { 192.168.0.0/24, 192.168.1.0/24 }
     19 
     20 #
     21 # Rules
     22 #
     23 
     24 # clear all rules
     25 flush ruleset
     26 
     27 table inet restricted {
     28 
     29     chain inbound {
     30 	type filter hook input priority 0;
     31 
     32 	    # allow established/related connections
     33 	    ct state {established, related} accept
     34 
     35 	    # allow localhost traffic
     36 	    iifname lo accept
     37 
     38 	    # echo requests (pings) 
     39 	    ip protocol icmp icmp type echo-request ct state new accept
     40 
     41 	    # reject everything else
     42 	    counter reject with icmp type port-unreachable
     43     }
     44 
     45     chain outbound {
     46 	type filter hook output priority 0;
     47 
     48 	    # allow localhost traffic
     49 	    oifname lo accept
     50 
     51 	    # allow VPN and exceptions directly
     52 	    oifname wlp2s0 ip daddr { $vpn_ip, $exceptions } accept
     53 
     54 	    # allow all traffic from UID tor
     55 	    oifname wlp2s0 skuid tor counter accept
     56 
     57 	    # allow wireguard traffic but only if destined to proxy
     58 	    oifname wg0 ip daddr { 10.64.0.1 } accept
     59 	    #oifname wg0 accept
     60 
     61 	    # reject remaining
     62 	    counter reject
     63     }
     64 
     65     # no forwarding
     66     chain forward {
     67 	type filter hook forward priority 0; policy drop;
     68     }
     69 }
